function renderChart() { const selectedEx = document.getElementById('chart-exercise-select').value; if (!selectedEx) return; const exDef = rawExercises.find(e => cleanString(e.name) === cleanString(selectedEx)); const exType = exDef ? (exDef.type || "Poids").trim().toLowerCase() : "poids"; const userLogs = journalLogs.filter(log => cleanString(log.exercise) === cleanString(selectedEx)); const dailyPerf = {}; let usedLest = false; userLogs.forEach(log => { const date = parseDateToJJMMAAAA(log.date); let w = parseFloat(String(log.weight).replace(/[^0-9.,-]/g, '').replace(',', '.')) || 0; let r = parseFloat(String(log.reps).replace(/[^0-9.,-]/g, '').replace(',', '.')) || 0; if (!dailyPerf[date]) dailyPerf[date] = { sum: 0, max_w: -Infinity, max_r: -Infinity, min_w: Infinity }; if (exType === "poids") { dailyPerf[date].sum += (w * r); if (w > dailyPerf[date].max_w) dailyPerf[date].max_w = w; } else if (exType === "pdc") { let lest = w > 0 ? w : 1; dailyPerf[date].sum += (r * lest); if (w > dailyPerf[date].max_w) dailyPerf[date].max_w = w; if (r > dailyPerf[date].max_r) dailyPerf[date].max_r = r; if (w > 0) usedLest = true; } else if (exType === "km") { dailyPerf[date].sum += w; if (w > dailyPerf[date].max_w) dailyPerf[date].max_w = w; } else if (exType === "durée" || exType === "duree") { dailyPerf[date].sum += w; if (w > 0 && w < dailyPerf[date].min_w) dailyPerf[date].min_w = w; } }); const dates = Object.keys(dailyPerf).sort((a, b) => { const [d1, m1, y1] = a.split('/'); const [d2, m2, y2] = b.split('/'); return new Date(y1, m1 - 1, d1) - new Date(y2, m2 - 1, d2); }); if (dates.length === 0) { if(chartInstance) chartInstance.destroy(); document.getElementById('no-data-msg').classList.remove('hidden'); document.getElementById('stats-details').classList.add('hidden'); return; } document.getElementById('no-data-msg').classList.add('hidden'); if (chartInstance) chartInstance.destroy(); const dataSum = dates.map(d => dailyPerf[d].sum); const dataMaxW = dates.map(d => dailyPerf[d].max_w !== -Infinity ? dailyPerf[d].max_w : 0); const dataMaxR = dates.map(d => dailyPerf[d].max_r !== -Infinity ? dailyPerf[d].max_r : 0); let datasets = []; if (exType === 'poids') { datasets.push( { label: 'Volume (Poids × Reps)', data: dataSum, yAxisID: 'y', borderColor: '#3b82f6', backgroundColor: 'rgba(59, 130, 246, 0.08)', borderWidth: 2.5, pointRadius: 3, pointHoverRadius: 6, pointBackgroundColor: '#3b82f6', fill: true, tension: 0.3 }, { label: 'Poids Max (kg)', data: dataMaxW, yAxisID: 'y1', borderColor: '#f59e0b', pointBackgroundColor: '#f59e0b', pointRadius: 3, pointHoverRadius: 6, borderWidth: 2, borderDash: [], fill: false, tension: 0.3 } ); } else if (exType === 'pdc') { datasets.push( { label: usedLest ? 'Score (Lest × Reps)' : 'Répétitions', data: dataSum, yAxisID: 'y', borderColor: '#10b981', backgroundColor: 'rgba(16, 185, 129, 0.08)', borderWidth: 2.5, pointRadius: 3, pointHoverRadius: 6, pointBackgroundColor: '#10b981', fill: true, tension: 0.3 }, { label: usedLest ? 'Lest Max (kg)' : 'Série Max', data: usedLest ? dataMaxW : dataMaxR, yAxisID: 'y1', borderColor: '#f59e0b', pointBackgroundColor: '#f59e0b', pointRadius: 3, pointHoverRadius: 6, borderWidth: 2, borderDash: [], fill: false, tension: 0.3 } ); } else { datasets.push({ label: 'Performance Max', data: dataMaxW, yAxisID: 'y', borderColor: '#8b5cf6', backgroundColor: 'rgba(139, 92, 246, 0.08)', borderWidth: 2.5, pointRadius: 3, pointHoverRadius: 6, pointBackgroundColor: '#8b5cf6', fill: true, tension: 0.3 }); } const ctx = document.getElementById('progressionChart').getContext('2d'); chartInstance = new Chart(ctx, { type: 'line', data: { labels: dates, datasets: datasets }, options: { responsive: true, maintainAspectRatio: false, layout: { padding: { top: 0 } }, plugins: { legend: { display: true, position: 'top', align: 'end', labels: { color: '#94a3b8', font: { size: 10, family: "'Inter', sans-serif", weight: '600' }, usePointStyle: false, boxWidth: 16, boxHeight: 2, padding: 8 } }, tooltip: { mode: 'index', intersect: false, backgroundColor: '#0f172a', borderColor: '#334155', borderWidth: 1, titleColor: '#f8fafc', bodyColor: '#cbd5e1' } }, scales: { x: { ticks: { color: '#64748b', font: { size: 10 }, maxTicksLimit: 5 }, grid: { display: false } }, y: { type: 'linear', display: true, position: 'left', ticks: { color: '#3b82f6', font: { size: 10, weight: '600' } }, grid: { display: false } // Desactive le quadrillage principal }, y1: { type: 'linear', display: datasets.length > 1, position: 'right', ticks: { color: '#f59e0b', font: { size: 10, weight: '600' } }, grid: { display: false } // Desactive le quadrillage secondaire } } } }); const statsDetails = document.getElementById('stats-details'); const topRecordsList = document.getElementById('top-records-list'); statsDetails.classList.remove('hidden'); let records = Object.keys(dailyPerf).map(date => ({ date, ...dailyPerf[date] })); records.sort((a, b) => b.max_w - a.max_w || b.max_r - a.max_r); topRecordsList.innerHTML = records.slice(0, 3).map((perf, index) => { const medalColors = ['text-amber-400', 'text-slate-300', 'text-amber-600']; return `
#${index + 1} ${perf.date}
${perf.max_w > 0 ? perf.max_w + 'kg' : perf.sum}
`; }).join(''); if (typeof lucide !== 'undefined') lucide.createIcons(); }
